#[derive(Debug, Clone)]
pub struct ProcessInfo {
    pub pid: u32,
    pub package_name: String,
    pub name: String,
}

pub fn parse_adb_ps_output(output: &str) -> Vec<ProcessInfo> {
    output.lines()
        .skip(1) // skip header
        .filter_map(|line| {
            let parts: Vec<&str> = line.split_whitespace().collect();
            if parts.len() >= 3 {
                let pid: u32 = parts[1].parse().ok()?;
                Some(ProcessInfo {
                    pid,
                    package_name: parts[2].to_string(),
                    name: parts.last().unwrap_or(&"").to_string(),
                })
            } else { None }
        })
        .collect()
}

impl AdbDevice {
    pub async fn list_processes(&self) -> Result<Vec<ProcessInfo>, String> {
        let output = self.shell("ps -A -o USER,PID,NAME").await?;
        Ok(parse_adb_ps_output(&output))
    }
}
```

// src-tauri/src/commands/cdp_server.rs

use tokio::net::TcpListener;
use tokio_tungstenite::accept_async;
use tokio_tungstenite::tungstenite::Message;

pub struct CdpServer {
    port: u16,
}

impl CdpServer {
    pub fn new(port: u16) -> Self { Self { port } }

    pub async fn start(&self) -> Result<(), String> {
        let addr = format!("127.0.0.1:{}", self.port);
        let listener = TcpListener::bind(&addr).await.map_err(|e| e.to_string())?;

        while let Ok((stream, _)) = listener.accept().await {
            let ws = accept_async(stream).await.map_err(|e| e.to_string())?;
            tokio::spawn(self.handle_connection(ws));
        }
        Ok(())
    }

    async fn handle_connection(&self, ws: WebSocketStream<TokioAdapter<net::TcpStream>>) {
        use futures_util::{SinkExt, StreamExt};
        let (mut write, mut read) = ws.split();
        while let Some(msg) = read.next().await {
            if let Ok(msg) = msg {
                let resp = self.handle_message(msg.into_data());
                if let Some(r) = resp {
                    let _ = write.send(Message::Text(r)).await;
                }
            }
        }
    }

    fn handle_message(&self, data: Vec<u8>) -> Option<String> {
        let msg: CdpMessage = serde_json::from_slice(&data).ok()?;
        self.dispatch(&msg).map(|r| serde_json::to_string(&r).unwrap_or_default())
    }

    fn dispatch(&self, msg: &CdpMessage) -> Option<CdpResponse> {
        let id = msg.id?;
        Some(CdpResponse {
            id,
            result: Some(serde_json::json!({})),
            error: None,
        })
    }
}
```
